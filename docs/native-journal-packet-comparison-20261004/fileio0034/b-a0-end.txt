	call Preload( "slot=1 arm=0 native-game-seconds=10.098" )
	call Preload( "source-method=generated-changing-no-file returned-string=none" )
	call Preload( "expected-read=none expected-wire=D0C+NNN+D0B+NNN+10zeroes" )
	call Preload( "attempts=300 sent=300 echoed=300 missing=0 bad-payload=0 local-read-errors=0 send-failures=0 duplicates=0 drain-callbacks=3" )
	call Preload( "read-count=0 native-read-seconds=0.000 sends-bytes=6600 send-call-seconds=0.000" )
	call Preload( "echo-age-samples=300 mean-native-seconds=0.105 max-native-seconds=0.223 phase-span-native-seconds=10.097" )

endfunction
