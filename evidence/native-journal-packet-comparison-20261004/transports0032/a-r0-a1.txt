	call Preload( "observer=0 rate=30 arm=1" )
	call Preload( "expected=120 received=120 missing=0" )
	call Preload( "bad=0 duplicates=0 send-failures=0" )
	call Preload( "own-echo-count=60 mean-native-seconds=0.080 max-native-seconds=0.099" )
	call Preload( "native-game-seconds=466.604 phase-seconds=2.100" )
	call PreloadEnd( 0.0 )

endfunction
